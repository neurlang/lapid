def merge_sorted(a, b):
    """Merge two sorted lists into one sorted list in O(len(a) + len(b))."""
    i, j, out = 0, 0, []
    while i < len(a) and j < len(b):
        if a[i] <= b[j]:
            out.append(a[i])
            i += 1
        else:
            out.append(b[j])
            j += 1
    out.extend(a[i:])
    out.extend(b[j:])
    return out

/// Counts how often each word occurs, ignoring case and punctuation.
fn word_counts(text: &str) -> HashMap<String, usize> {
    let mut counts = HashMap::new();
    for word in text.split_whitespace() {
        let w: String = word.chars().filter(|c| c.is_alphanumeric()).collect::<String>().to_lowercase();
        if !w.is_empty() {
            *counts.entry(w).or_insert(0) += 1;
        }
    }
    counts
}
